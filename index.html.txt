    <script>
        let currentSport = 'cricket';
        let state = {
            cricket: { runs: 0, wickets: 0, balls: 0 },
            badminton: { s1: 0, s2: 0, sets1: 0, sets2: 0 },
            carrom: { s1: 0, s2: 0 },
            chess: { status: 'In Progress...' }
        };

        if (localStorage.getItem('tournament_state')) {
            state = JSON.parse(localStorage.getItem('tournament_state'));
            updateUI();
        }

        function switchSport(sport) {
            currentSport = sport;
            document.querySelectorAll('.sport-panel').forEach(p => p.classList.add('hidden'));
            document.getElementById(`sport-${sport}`).classList.remove('hidden');

            document.querySelectorAll('.tab-btn').forEach(b => {
                b.classList.remove('bg-amber-500', 'text-slate-950');
                b.classList.add('text-slate-400');
            });
            const activeTab = document.getElementById(`tab-${sport}`);
            activeTab.classList.add('bg-amber-500', 'text-slate-950');
            activeTab.classList.remove('text-slate-400');
        }

        function saveState() {
            localStorage.setItem('tournament_state', JSON.stringify(state));
        }

        function updateUI() {
            document.getElementById('cricket-runs').innerText = state.cricket.runs;
            document.getElementById('cricket-wickets').innerText = state.cricket.wickets;
            const overs = Math.floor(state.cricket.balls / 6) + '.' + (state.cricket.balls % 6);
            document.getElementById('cricket-overs').innerText = overs;

            document.getElementById('badminton-s1').innerText = state.badminton.s1;
            document.getElementById('badminton-s2').innerText = state.badminton.s2;
            document.getElementById('badminton-sets1').innerText = state.badminton.sets1;
            document.getElementById('badminton-sets2').innerText = state.badminton.sets2;

            document.getElementById('carrom-s1').innerText = state.carrom.s1;
            document.getElementById('carrom-s2').innerText = state.carrom.s2;

            document.getElementById('chess-status').innerText = state.chess.status;

            saveState();
        }

        function addRun(runs) {
            state.cricket.runs += runs;
            state.cricket.balls += 1;
            updateUI();
        }
        function addExtra(type) {
            state.cricket.runs += 1;
            updateUI();
        }
        function addWicket() {
            if (state.cricket.wickets < 10) {
                state.cricket.wickets += 1;
                state.cricket.balls += 1;
                updateUI();
            }
        }

        function badmintonPoint(player) {
            if (player === 1) state.badminton.s1++;
            else state.badminton.s2++;

            if (state.badminton.s1 >= 21 && (state.badminton.s1 - state.badminton.s2 >= 2)) {
                state.badminton.sets1++;
                state.badminton.s1 = 0; state.badminton.s2 = 0;
            } else if (state.badminton.s2 >= 21 && (state.badminton.s2 - state.badminton.s1 >= 2)) {
                state.badminton.sets2++;
                state.badminton.s1 = 0; state.badminton.s2 = 0;
            }
            updateUI();
        }

        function carromAdd(player, pts) {
            if (player === 1) state.carrom.s1 += pts;
            else state.carrom.s2 += pts;
            updateUI();
        }

        function chessResult(res) {
            state.chess.status = `Finished: ${res}`;
            updateUI();
        }

        function resetCurrentSport() {
            if (confirm(`Reset match data for ${currentSport.toUpperCase()}?`)) {
                if (currentSport === 'cricket') state.cricket = { runs: 0, wickets: 0, balls: 0 };
                if (currentSport === 'badminton') state.badminton = { s1: 0, s2: 0, sets1: 0, sets2: 0 };
                if (currentSport === 'carrom') state.carrom = { s1: 0, s2: 0 };
                if (currentSport === 'chess') state.chess.status = 'In Progress...';
                updateUI();
            }
        }
    </script>
